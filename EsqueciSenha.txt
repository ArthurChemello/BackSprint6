  email TEXT NOT NULL,
  code TEXT NOT NULL,
  expires_at TIMESTAMP WITH TIME ZONE NOT NULL,
  used BOOLEAN DEFAULT false,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()

async forgotPassword(email: string) {
  // Verifica se o email existe em doctors ou patients
  const { data: doctor } = await this.supabaseService.supabase
    .from('doctors')
    .select('id, email')
    .eq('email', email)
    .single();

  const { data: patient } = await this.supabaseService.supabase
    .from('patients')
    .select('id, email')
    .eq('email', email)
    .single();

  if (!doctor && !patient) {
    throw new Error('Email não encontrado');
  }

  // Gera código de 6 caracteres
  const code = Math.random().toString(36).substring(2, 8).toUpperCase();
  const expires_at = new Date(Date.now() + 60 * 60 * 1000); // 1 hora

  // Salva no banco
  await this.supabaseService.supabase
    .from('password_resets')
    .insert({ email, code, expires_at });

  // Envia email
  await this.mailService.sendPasswordReset(email, code);

  return { message: 'Código enviado para o email!' };
}

async resetPassword(email: string, code: string, newPassword: string) {
  // Verifica o código
  const { data: reset } = await this.supabaseService.supabase
    .from('password_resets')
    .select('*')
    .eq('email', email)
    .eq('code', code)
    .eq('used', false)
    .single();

  if (!reset) throw new Error('Código inválido');

  if (new Date() > new Date(reset.expires_at)) {
    throw new Error('Código expirado');
  }

  const hashedPassword = await bcrypt.hash(newPassword, 10);

  // Atualiza senha em doctors
  await this.supabaseService.supabase
    .from('doctors')
    .update({ password: hashedPassword })
    .eq('email', email);

  // Atualiza senha em patients
  await this.supabaseService.supabase
    .from('patients')
    .update({ password: hashedPassword, first_login: false })
    .eq('email', email);

  // Marca código como usado
  await this.supabaseService.supabase
    .from('password_resets')
    .update({ used: true })
    .eq('id', reset.id);

  return { message: 'Senha alterada com sucesso!' };
}
